# Session Intelligence — Database Schema & Data Contracts

This document specifies the database tables, JSON structures, and data contract between **Part 1 (Google Apps Script / Gemini Transcriber)** and **Part 2 (Admin Interactive UI & Conversational Chat)**.

---

## 1. Table: `session_transcripts`

Stores the raw speech-to-text transcript along with structured clinical insights extracted by Gemini.

```sql
CREATE TABLE IF NOT EXISTS public.session_transcripts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  booking_id UUID REFERENCES public.bookings(id) ON DELETE SET NULL,
  client_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL,
  client_name TEXT NOT NULL,
  client_email TEXT,
  session_date TIMESTAMPTZ NOT NULL,
  duration_minutes INTEGER NOT NULL DEFAULT 50,
  google_meet_url TEXT,
  focus_areas TEXT[] DEFAULT '{}'::TEXT[],
  
  -- Structured clinical intelligence (generated by Gemini)
  clinical_summary TEXT NOT NULL,
  key_insights TEXT[] DEFAULT '{}'::TEXT[],
  action_items JSONB NOT NULL DEFAULT '[]'::JSONB,
  emotional_observations JSONB DEFAULT '{}'::JSONB,
  
  -- Raw dialogue transcript
  raw_transcript TEXT NOT NULL,
  
  -- Processing lifecycle
  status TEXT NOT NULL DEFAULT 'completed' CHECK (status IN ('processing', 'completed', 'failed')),
  metadata JSONB DEFAULT '{}'::JSONB,
  
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Indexes for rapid lookup
CREATE INDEX IF NOT EXISTS idx_session_transcripts_client_id ON public.session_transcripts(client_id);
CREATE INDEX IF NOT EXISTS idx_session_transcripts_booking_id ON public.session_transcripts(booking_id);
CREATE INDEX IF NOT EXISTS idx_session_transcripts_session_date ON public.session_transcripts(session_date DESC);

-- RLS: Only admins can view or modify session transcripts
ALTER TABLE public.session_transcripts ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Admins full access to session transcripts"
  ON public.session_transcripts
  FOR ALL
  TO authenticated
  USING (
    EXISTS (
      SELECT 1 FROM public.profiles
      WHERE profiles.id = auth.uid()
      AND profiles.role = 'admin'
    )
  );
```

## 2. Table: `session_chat_messages`

Persists multi-turn context-aware conversations between Mai and Gemini regarding a client or session.

```sql
CREATE TABLE IF NOT EXISTS public.session_chat_messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID REFERENCES public.session_transcripts(id) ON DELETE CASCADE,
  client_id TEXT NOT NULL,
  sender TEXT NOT NULL CHECK (sender IN ('admin', 'gemini')),
  content TEXT NOT NULL,
  metadata JSONB DEFAULT '{}'::JSONB,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_session_chat_session_id ON public.session_chat_messages(session_id);
CREATE INDEX IF NOT EXISTS idx_session_chat_created_at ON public.session_chat_messages(created_at ASC);

ALTER TABLE public.session_chat_messages ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Admins full access to session chat messages"
  ON public.session_chat_messages
  FOR ALL
  TO authenticated
  USING (
    EXISTS (
      SELECT 1 FROM public.profiles
      WHERE profiles.id = auth.uid()
      AND profiles.role = 'admin'
    )
  );
```
